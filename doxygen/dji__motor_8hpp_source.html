67 static void SendCommand() {
68 for (auto &[can, buffers] : tx_buf_) {
69 if (buffers.dirty_200) {
70 can->Write(0x200, buffers.data_200, 8);
71 buffers.dirty_200 = false;
72 }
73 if (buffers.dirty_1ff) {
74 can->Write(0x1ff, buffers.data_1ff, 8);
75 buffers.dirty_1ff = false;
76 }
77 if (buffers.dirty_2ff) {
78 can->Write(0x2ff, buffers.data_2ff, 8);
79 buffers.dirty_2ff = false;
80 }
81 }
82 }

88 static void SendCommand(hal::CanInterface &can) {
89 auto target_can = tx_buf_.find(&can);
90 if (target_can == tx_buf_.end()) {
91 return;
92 }
93 auto &buffers = target_can->second;
94
95 if (buffers.dirty_200) {
96 can.Write(0x200, buffers.data_200, 8);
97 buffers.dirty_200 = false;
98 }
99 if (buffers.dirty_1ff) {
100 can.Write(0x1ff, buffers.data_1ff, 8);
101 buffers.dirty_1ff = false;
102 }
103 if (buffers.dirty_2ff) {
104 can.Write(0x2ff, buffers.data_2ff, 8);
105 buffers.dirty_2ff = false;
106 }
107 }

181 DjiMotor(hal::CanInterface &can, u16 id, bool reversed = false)
182 : CanDevice(can, GetRxIdBase() + id), id_(id), reversed_(reversed) {
183 // 检查id是否合法
184 if (id_ < 1 || id_ > 8) {
185 rm::Throw(std::runtime_error("DjiMotor: invalid motor id"));
186 }
187 if constexpr (motor_type == DjiMotorType::kGM6020) {
188 if (id_ > 7) { // 6020没有8号
189 rm::Throw(std::runtime_error("DjiMotor: invalid motor id for GM6020"));
190 }
191 }
192 // 如果这个电机所属的CAN总线还没有对应的发送缓冲区，就创建一个
193 if (tx_buf_.find(&can) == tx_buf_.end()) {
194 tx_buf_.insert({&can, TxBuffers{}});
195 }
196 }

203 void SetCurrent(i16 current) {
204 // 限幅到电机能接受的最大电流
205 current = modules::Clamp(current, -GetCurrentBound(), GetCurrentBound());
206 // 处理反转
207 if (this->reversed_) {
208 current = -current;
209 }
210 // 找到这个电机所属的CAN总线的发送缓冲区
211 auto buffers = tx_buf_.find(this->can_);
212 if (buffers == tx_buf_.end()) {
213 // 这个电机没有对应的CAN总线缓冲区，理论上不会发生，但是为安全起见还是加上这个判断
214 return;
215 }
216 auto &[_, buf] = *buffers;
217 // 根据电机型号和ID(this->id_)找到对应的发送缓冲区，写入电流值，标记缓冲区已修改
218 if constexpr (motor_type == DjiMotorType::kGM6020) {
219 if (1 <= this->id_ && this->id_ <= 4) {
220 buf.data_1ff[(this->id_ - 1) * 2] = (current >> 8) & 0xff;
221 buf.data_1ff[(this->id_ - 1) * 2 + 1] = current & 0xff;
222 buf.dirty_1ff = true;
223 } else if (5 <= this->id_ && this->id_ <= 8) {
224 buf.data_2ff[(this->id_ - 5) * 2] = (current >> 8) & 0xff;
225 buf.data_2ff[(this->id_ - 5) * 2 + 1] = current & 0xff;
226 buf.dirty_2ff = true;
227 }
228 } else if constexpr (motor_type == DjiMotorType::kM3508 || //
229 motor_type == DjiMotorType::kM2006) {
230 if (1 <= this->id_ && this->id_ <= 4) {
231 buf.data_200[(this->id_ - 1) * 2] = (current >> 8) & 0xff;
232 buf.data_200[(this->id_ - 1) * 2 + 1] = current & 0xff;
233 buf.dirty_200 = true;
234 } else if (5 <= this->id_ && this->id_ <= 8) {
235 buf.data_1ff[(this->id_ - 5) * 2] = (current >> 8) & 0xff;
236 buf.data_1ff[(this->id_ - 5) * 2 + 1] = current & 0xff;
237 buf.dirty_1ff = true;
238 }
239 }
240 }
