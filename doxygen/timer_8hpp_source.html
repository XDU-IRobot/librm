49class PollingTimer : public TimerInterface {
50 public:
54 PollingTimer()
55 : running_(false),
56 timeout_ms_(0),
57 elapsed_ms_(0),
58 callback_{},
59 type_(TimerType::kPeriodic),
60 timeout_set_(false) {}
61
65 ~PollingTimer() override = default;
66
71 bool Start() override {
72 if (!timeout_set_) {
73 return false; // 还没有成功设置过超时时间，启动失败
74 }
75 elapsed_ms_ = 0;
76 running_ = true;
77 return true;
78 }
79
84 bool Stop() override {
85 if (!running_) {
86 return false;
87 }
88 running_ = false;
89 return true;
90 }
91
96 bool running() const override { return running_; }
97
103 bool SetTimeout(std::chrono::duration<f32> timeout) override {
104 if (running_) {
105 return false;
106 }
107 if (timeout <= std::chrono::duration<f32>::zero()) {
108 return false;
109 }
110 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(timeout).count();
111 if (ms < 1) {
112 ms = 1;
113 }
114 timeout_ms_ = static_cast<u32>(ms);
115 timeout_set_ = true;
116 return true;
117 }
118
124 std::chrono::duration<f32> GetTimeout() const override {
125 if (!timeout_set_) {
126 return std::chrono::duration<f32>::zero();
127 }
128 return std::chrono::duration<f32>(static_cast<f32>(timeout_ms_) / 1000.0f);
129 }
130
136 bool AttachCallback(etl::delegate<void()> callback) override {
137 if (running_) {
138 return false;
139 }
140 callback_ = callback;
141 return true;
142 }
143
151 bool SetTimerType(TimerType type) override {
152 if (running_) {
153 return false;
154 }
155 type_ = type;
156 return true;
157 }
158
165 void Poll(uint32_t time_increment_ms) {
166 if (!running_) return;
167 elapsed_ms_ += time_increment_ms;
168 if (elapsed_ms_ >= timeout_ms_) {
169 if (type_ == TimerType::kOneShot) {
170 running_ = false;
171 } else {
172 elapsed_ms_ = 0;
173 }
174 if (callback_) {
175 callback_();
176 }
177 }
178 }
179
180 private:
181 bool running_;
182 uint32_t timeout_ms_;
183 uint32_t elapsed_ms_;
184 etl::delegate<void()> callback_;
185 TimerType type_;
186 bool timeout_set_;
187};
