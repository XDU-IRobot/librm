83 bool Write(u16 id, const u8 *data, usize size, u8 priority, time_point deadline) {
84 TxRequest req;
85 req.id = id;
86 req.size = (size <= 64) ? size : 64;
87 std::copy_n(data, req.size, req.data.begin());
88 return queue_.Push(req, priority, deadline);
89 }
90
96 bool Process() {
97 auto result = queue_.Process();
98 if (result.has_value()) {
99 const auto &req = result.value();
100 FdCan::Write(req.id, req.data.data(), req.size);
101 return true;
102 }
103 return false;
104 }
105
111 bool Process(time_point now) {
112 auto result = queue_.Process(now);
113 if (result.has_value()) {
114 const auto &req = result.value();
115 FdCan::Write(req.id, req.data.data(), req.size);
116 return true;
117 }
118 return false;
119 }
120
124 void Stop() override {
125 FdCan::Stop();
126 queue_.Clear();
127 }

129 const auto &queue() const { return queue_; }
