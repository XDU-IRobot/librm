136 bool Process(time_point now) {
137 // 采样队列深度
138 const usize depth = queue_.size();
139 depth_avg_.add(static_cast<f32>(depth));
140 if (depth > stat_window_data_.peak_depth) {
141 stat_window_data_.peak_depth = depth;
142 }
143
144 auto result = queue_.Process(now);
145 const bool sent = result.has_value();
146 if (sent) {
147 const auto &req = result.value();
148 // 把队列项还原成CanFrame，帧类型按入队时记录的值显式透传
149 CanFrame frame{};
150 frame.std_id = req.id;
151 frame.dlc = static_cast<CanFrame::Dlc>(req.size);
152 frame.is_fd_frame = req.is_fd_frame;
153 std::copy_n(req.data.begin(), req.size, frame.data.begin());
154 Base::Write(frame);
155 ++stat_window_data_.tx_count;
156 }
157
158 // 检查是否到了 1 秒刷新窗口
159 if (now - stats_window_start_ >= std::chrono::seconds(1)) {
160 const f32 elapsed_s = std::chrono::duration<f32>(now - stats_window_start_).count();
161
162 const usize expired_delta = queue_.expired_count() - stat_window_data_.expired_baseline;
163
164 // clang-format off
165 TxStats snap;
166 snap.tx_fps = static_cast<f32>(stat_window_data_.tx_count) / elapsed_s;
167 snap.enqueue_fps = static_cast<f32>(stat_window_data_.enqueue_count) / elapsed_s;
168 snap.drop_full_fps = static_cast<f32>(stat_window_data_.drop_full_count) / elapsed_s;
169 snap.drop_expired_fps = static_cast<f32>(expired_delta) / elapsed_s;
170 snap.drop_total_fps = snap.drop_full_fps + snap.drop_expired_fps;
171 snap.peak_queue_depth = stat_window_data_.peak_depth;
172 snap.avg_queue_depth = depth_avg_.value();
173 stats_snapshot_ = snap;
174
175 // 重置窗口
176 stats_window_start_ = now;
177 stat_window_data_.tx_count = 0;
178 stat_window_data_.enqueue_count = 0;
179 stat_window_data_.drop_full_count = 0;
180 stat_window_data_.expired_baseline = queue_.expired_count();
181 stat_window_data_.peak_depth = 0;
182 depth_avg_.clear(0.0f);
183 // clang-format on
184 }
185
186 return sent;
187 }

217 private:
228 bool Enqueue(u16 id, const u8 *data, usize size, bool is_fd_frame, u8 priority, time_point deadline) {
229 // 先校验再入队，保证错误配置不会被静默丢弃或截断
230 if (!CanFrame::IsLegalLength(size)) {
231 Throw(rm::runtime_error("throttled_can: dlc is not a legal CAN frame length"));
232 }
233 if (size > MaxDataSize) {
234 Throw(rm::runtime_error("throttled_can: frame is too long for this queue"));
235 }
236 if (is_fd_frame && !kSupportsFdFrame) {
237 Throw(rm::runtime_error("throttled_can: CAN FD frame requested, but the underlying CAN does not support CAN FD"));
238 }
239
240 TxRequest req;
241 req.id = id;
242 req.size = size;
243 req.is_fd_frame = is_fd_frame;
244 std::copy_n(data, size, req.data.begin());
245 ++stat_window_data_.enqueue_count;
246 if (!queue_.Push(req, priority, deadline)) {
247 ++stat_window_data_.drop_full_count;
248 return false;
249 }
250 return true;
251 }

253 struct TxRequest {
254 u16 id;
255 std::array<u8, MaxDataSize> data;
256 usize size;
257 bool is_fd_frame{false};
258 };

263

270 etl::pseudo_moving_average<f32, 16> depth_avg_{0.0f};
271

273 struct {
274 usize tx_count{0};
275 usize enqueue_count{0};
276 usize drop_full_count{0};
277 usize expired_baseline{0};
278 usize peak_depth{0};
279 } stat_window_data_;
280};
