46class M3508PowerModel {
47 public:
48 // 电机参数常量
49 static constexpr f32 kKt = 0.3f / (3591.0f / 187.0f);
50 static constexpr f32 kKe = (60.0f / (2.0f * M_PI * 24.48f)) / (3591.0f / 187.0f);
51 static constexpr f32 kTorqueCoeff = 1.99688994e-6f;
52
53 // 拟合模型系数
54 static constexpr f32 kK1 = 1.453e-07f;
55 static constexpr f32 kK2 = 1.23e-07f;
56 static constexpr f32 kConst = 4.081f;
57
58 static constexpr f32 kCurrentToTorqueCurrent = 20.0f / 16384.0f;
59 static constexpr f32 kSpeedToMechanicalPower = 1.0f / 9.55f;
60 static constexpr i16 kMaxCurrent = 16000;
61
65 struct MotorState {
66 f32 speed_rpm;
67 f32 give_current;
68 f32 measured_current;
69 };

74 struct PowerInfo {
75 f32 mechanical_power;
76 f32 loss_power;
77 f32 total_power;
78 f32 torque;
79 };

88 PowerInfo CalculatePower(const MotorState& state) const {
89 PowerInfo info;
90
91 // 计算输出转矩: τ = KT * I_torque
92 f32 torque_current = state.give_current * kCurrentToTorqueCurrent;
93 info.torque = kKt * torque_current;
94
95 // 计算机械功率: P_mech = τ·ω / 9.55
96 info.mechanical_power = (info.torque * state.speed_rpm) * kSpeedToMechanicalPower;
97
98 // 计算损耗功率: P_loss = k1·ω² + k2·I² + C
99 f32 speed_squared = state.speed_rpm * state.speed_rpm;
100 f32 current_squared = state.give_current * state.give_current;
101 info.loss_power = kK1 * speed_squared + kK2 * current_squared + kConst;
102
103 // 总功率
104 info.total_power = info.mechanical_power + info.loss_power;
105
106 return info;
107 }

116 f32 CalculateCtrlForPower(f32 target_power, f32 speed_rpm, bool is_positive_direction) const {
117 // 求解二次方程: a·I² + b·I + c = 0
118 // 其中功率模型为: P = (KT·I_torque)·ω/9.55 + k1·ω² + k2·I² + C
119 // P = TORQUE_COEFFICIENT·I·ω + k1·ω² + k2·I² + C
120
121 f32 a = kK2;
122 f32 b = kTorqueCoeff * speed_rpm;
123 f32 c = kK1 * speed_rpm * speed_rpm - target_power + kConst;
124
125 // 判别式
126 f32 discriminant = b * b - 4.0f * a * c;
127
128 if (discriminant < 0) {
129 // 无实数解，目标功率无法达到
130 return 0.0f;
131 }
132
133 f32 sqrt_discriminant = std::sqrt(discriminant);
134 f32 current;
135
136 // 根据转矩方向选择解
137 if (is_positive_direction) {
138 current = (-b + sqrt_discriminant) / (2.0f * a);
139 } else {
140 current = (-b - sqrt_discriminant) / (2.0f * a);
141 }
142
143 // 限幅
144 if (current > kMaxCurrent) {
145 current = kMaxCurrent;
146 } else if (current < -kMaxCurrent) {
147 current = -kMaxCurrent;
148 }
149
150 return current;
151 }

160 template <usize NMotors>
161 void DistributePower(const std::array<MotorState, NMotors> motor_states, const f32* initial_currents,
162 f32 max_total_power, f32* output_currents) const {
163 // 计算初始总功率
164 f32 initial_total_power = 0.0f;
165 f32 initial_powers[NMotors];
166
167 for (size_t i = 0; i < NMotors; ++i) {
168 MotorState state = motor_states[i];
169 state.give_current = initial_currents[i];
170
171 const PowerInfo info = CalculatePower(state);
172 initial_powers[i] = info.total_power;
173
174 // 负功率不计入（过渡状态）
175 if (initial_powers[i] > 0) {
176 initial_total_power += initial_powers[i];
177 }
178 }
179
180 // 如果超过最大功率，按比例缩放
181 if (initial_total_power > max_total_power) {
182 f32 power_scale = max_total_power / initial_total_power;
183
184 for (size_t i = 0; i < NMotors; ++i) {
185 f32 scaled_power = initial_powers[i] * power_scale;
186
187 if (scaled_power < 0) {
188 output_currents[i] = initial_currents[i];
189 continue;
190 }
191
192 // 根据缩放后的功率重新计算电流
193 bool is_positive = initial_currents[i] > 0;
194 output_currents[i] = CalculateCtrlForPower(scaled_power, motor_states[i].speed_rpm, is_positive);
195 }
196 } else {
197 // 不超过功率限制，直接使用初始电流
198 for (size_t i = 0; i < NMotors; ++i) {
199 output_currents[i] = initial_currents[i];
200 }
201 }
202 }
203};
