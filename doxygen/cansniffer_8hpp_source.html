73template <usize MaxIds = 32, usize WindowSize = 16, usize MaxPendingFrames = 32>
74class CanSniffer : public hal::CanFrameObserver {
75 static_assert(MaxIds > 0, "MaxIds must be greater than 0");
76 static_assert(WindowSize > 0, "WindowSize must be greater than 0");
77 static_assert(MaxPendingFrames > 0, "MaxPendingFrames must be greater than 0");

99 void Update() {
100 while (true) {
101 const u32 tail = tail_.load(std::memory_order_relaxed);
102 if (tail == head_.load(std::memory_order_acquire)) {
103 break;
104 }
105 const PendingFrame frame = pending_[tail];
106 tail_.store((tail + 1) % kPendingCapacity, std::memory_order_release);
107 Accumulate(frame);
108 }
109 }

115 [[nodiscard]] StatsList stats() const {
116 StatsList result;
117 for (const auto &entry : entries_) {
118 result.push_back(MakeStats(entry));
119 }
120 return result;

128 [[nodiscard]] std::optional<CanIdStats> stats(u16 id) const {
129 for (const auto &entry : entries_) {

142 [[nodiscard]] IdList active_ids(duration timeout = std::chrono::seconds(1)) const {
143 IdList ids;
144 const u32 now = CurrentTimeUs();
145 const u32 timeout_us = TimeoutToUs(timeout);
146 for (const auto &entry : entries_) {
147 if (entry.frame_count > 0 && (now - entry.last_timestamp_us) <= timeout_us) {
148 ids.push_back(entry.id);
149 }
150 }
151 return ids;
152 }
153
158 [[nodiscard]] u32 dropped_frames() const { return dropped_frames_.load(std::memory_order_relaxed); }
159
164 void notification(const hal::CanFrame &msg) override {
165 const u32 head = head_.load(std::memory_order_relaxed);
166 const u32 next = (head + 1) % kPendingCapacity;
167 if (next == tail_.load(std::memory_order_acquire)) {
168 dropped_frames_.fetch_add(1, std::memory_order_relaxed);
169 return;
170 }
171
172 PendingFrame &frame = pending_[head];
173 frame.timestamp_us = CurrentTimeUs();
174 frame.id = msg.std_id;
175 frame.dlc = static_cast<u8>(msg.dlc);
176 frame.is_fd_frame = msg.is_fd_frame;
177 head_.store(next, std::memory_order_release);
178 }
179
180 private:
181 static constexpr usize kPendingCapacity = MaxPendingFrames + 1;

183 struct PendingFrame {
184 u32 timestamp_us{0};
185 u16 id{0};
186 u8 dlc{0};
187 bool is_fd_frame{false};
188 };

190 struct Entry {
191 u16 id{0};
192 u8 dlc{0};
193 bool is_fd_frame{false};
194 u32 frame_count{0};
195 u32 last_timestamp_us{0};
196 usize sample_count{0};
197 usize next_index{0};
198 std::array<u32, WindowSize> timestamps_us{};
199 };
200
207 void Accumulate(const PendingFrame &frame) {
208 Entry *entry = FindOrCreate(frame.id);
209 if (entry == nullptr) {
210 dropped_frames_.fetch_add(1, std::memory_order_relaxed);
211 return;
212 }
213
214 if (entry->frame_count == 0) {
215 PushTimestamp(*entry, frame.timestamp_us);
216 entry->last_timestamp_us = frame.timestamp_us;
217 } else if (static_cast<i32>(frame.timestamp_us - entry->last_timestamp_us) > 0) {
218 PushTimestamp(*entry, frame.timestamp_us);
219 entry->last_timestamp_us = frame.timestamp_us;
220 }
221
222 entry->dlc = frame.dlc;
223 entry->is_fd_frame = frame.is_fd_frame;
224 ++entry->frame_count;
225 }
226
231 Entry *FindOrCreate(u16 id) {
232 usize pos = 0;
233 while (pos < entries_.size() && entries_[pos].id < id) {
234 ++pos;
235 }
236 if (pos < entries_.size() && entries_[pos].id == id) {
237 return &entries_[pos];
238 }
239 if (entries_.full()) {
240 return nullptr;
241 }
242
243 entries_.push_back(Entry{});
244 for (usize i = entries_.size() - 1; i > pos; --i) {
245 entries_[i] = entries_[i - 1];
246 }
247 entries_[pos].id = id;
248 return &entries_[pos];
249 }

254 static u32 TimestampAt(const Entry &entry, usize index) {
255 if (entry.sample_count < WindowSize) {
256 return entry.timestamps_us[index];
257 }
258 return entry.timestamps_us[(entry.next_index + index) % WindowSize];
259 }
260
264 static void PushTimestamp(Entry &entry, u32 timestamp_us) {
265 entry.timestamps_us[entry.next_index] = timestamp_us;
266 entry.next_index = (entry.next_index + 1) % WindowSize;
267 if (entry.sample_count < WindowSize) {
268 ++entry.sample_count;
269 }
270 }
271
277 static CanIdStats MakeStats(const Entry &entry) {
278 CanIdStats result;
279 result.id = entry.id;
280 result.frame_count = entry.frame_count;
281 result.dlc = entry.dlc;
282 result.is_fd_frame = entry.is_fd_frame;
283 if (entry.sample_count < 2) {
284 return result;
285 }
286
287 const usize count = entry.sample_count;
288 const u32 span_us = TimestampAt(entry, count - 1) - TimestampAt(entry, 0);
289 if (span_us == 0) {
290 return result;
291 }
292
293 const f32 mean_period_us = static_cast<f32>(span_us) / static_cast<f32>(count - 1);
294 result.average_hz = static_cast<f32>(count - 1) * 1e6f / static_cast<f32>(span_us);
295
296 f32 squared_deviation_sum = 0.0f;
297 for (usize i = 1; i < count; ++i) {
298 const f32 period_us = static_cast<f32>(TimestampAt(entry, i) - TimestampAt(entry, i - 1));
299 const f32 deviation = period_us - mean_period_us;
300 squared_deviation_sum += deviation * deviation;
301 }
302 const f32 stddev_period_us = std::sqrt(squared_deviation_sum / static_cast<f32>(count - 1));
303 result.stddev_hz = stddev_period_us * 1e6f / (mean_period_us * mean_period_us);
304 return result;
305 }
306
310 static u32 CurrentTimeUs() {
311 return static_cast<u32>(
312 std::chrono::duration_cast<std::chrono::microseconds>(clock::now().time_since_epoch()).count());
313 }
314
318 static u32 TimeoutToUs(duration timeout) {
319 if (timeout <= duration::zero()) {
320 return 0;
321 }
322 const auto timeout_us = std::chrono::duration_cast<std::chrono::microseconds>(timeout).count();
323 return timeout_us > static_cast<i64>(0xFFFFFFFFu) ? 0xFFFFFFFFu : static_cast<u32>(timeout_us);
324 }
