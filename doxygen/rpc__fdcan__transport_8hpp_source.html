79 void Send(const uint8_t *data, uint32_t size) {
80 // 更新最后发送时间，数据帧本身就能维持连接
81 last_tx_time_ = std::chrono::steady_clock::now();
82
83 // 1. 生成一个随机的帧ID，用于标识本次传输的所有分片
84 tx_buf_.frame_start.frame_id = static_cast<uint8_t>(rng_() & 0xff);
85
86 // 2. 准备起始帧，包含元数据：帧ID、总载荷长度和CRC8校验值
87 tx_buf_.frame_start.payload_len = size;
88 tx_buf_.frame_start.payload_crc8 = modules::Crc8(data, size, modules::CRC8_INIT);
89 // 发送起始帧
90 can_->Write(tx_stdid_base_ + StdIdOffset::kFrameStart, reinterpret_cast<const uint8_t *>(&tx_buf_.frame_start),
91 sizeof(tx_buf_.frame_start));
92
93 // 3. 循环发送数据分片
94 uint16_t fragment_seq = 0;
95 uint32_t bytes_sent = 0;
96 while (bytes_sent < size) {
97 tx_buf_.payload.frame_id = tx_buf_.frame_start.frame_id;
98 tx_buf_.payload.fragment_seq = fragment_seq++;
99 // 计算当前分片的数据大小
100 uint8_t fragment_size =
101 std::min(static_cast<uint32_t>(sizeof(tx_buf_.payload.fragment_data)), size - bytes_sent);
102 tx_buf_.payload.fragment_len = fragment_size;
103 std::memcpy(tx_buf_.payload.fragment_data, data + bytes_sent, fragment_size);
104 // 发送数据分片，DLC固定为64
105 can_->Write(tx_stdid_base_ + StdIdOffset::kPayload, reinterpret_cast<const uint8_t *>(&tx_buf_.payload), 64);
106 bytes_sent += fragment_size;
107 }
108
109 // 4. 发送结束帧，标识数据传输完成
110 tx_buf_.frame_end.frame_id = tx_buf_.frame_start.frame_id;
111 can_->Write(tx_stdid_base_ + StdIdOffset::kFrameEnd, reinterpret_cast<const uint8_t *>(&tx_buf_.frame_end),
112 sizeof(tx_buf_.frame_end));
113 }

121 void SendHeartbeatIfNeeded() {
122 auto now = std::chrono::steady_clock::now();
123 auto elapsed_ms = std::chrono::duration_cast<std::chrono::milliseconds>(now - last_tx_time_).count();
124 if (elapsed_ms >= kHeartbeatIntervalMs) {
125 can_->Write(tx_stdid_base_ + StdIdOffset::kHeartbeat, nullptr, 0);
126 last_tx_time_ = now;
127 }

133 void RxCallback(const hal::CanFrame *msg) override {
134 // 只处理本通道的4种帧，其余报文一律忽略（否则总线上其他设备的报文会被误认为对方在线）
135 const uint16_t id = msg->std_id;
136 const bool is_own_frame =
137 (id == rx_stdid_base_ + StdIdOffset::kHeartbeat) || (id == rx_stdid_base_ + StdIdOffset::kFrameStart) ||
138 (id == rx_stdid_base_ + StdIdOffset::kPayload) || (id == rx_stdid_base_ + StdIdOffset::kFrameEnd);
139 if (!is_own_frame) {
140 return;
141 }
142
143 // 收到本通道的任何帧都更新连接状态，因为任何帧都能证明对方在线
144 ReportStatus(kOk);
145
146 // 判断帧类型
147 if (id == rx_stdid_base_ + StdIdOffset::kHeartbeat) {
148 // 心跳帧，已在上面更新状态
149 } else if (msg->std_id == rx_stdid_base_ + StdIdOffset::kFrameStart) {
150 // 1. 收到起始帧，开始一个新的数据帧接收会话
151 const auto *frame_start = reinterpret_cast<const decltype(tx_buf_.frame_start) *>(msg->data.data());
152 if (static_cast<usize>(msg->dlc) != sizeof(*frame_start)) {
153 return; // 帧长度错误，直接丢弃
154 }
155 if (frame_start->payload_len > kMaxPayloadBytes) {
156 // 载荷超出单帧允许的最大长度，直接丢弃该会话
157 rx_sessions_.erase(frame_start->frame_id);
158 return;
159 }
160 // 检查会话池是否已满，如果满了且是新会话，则淘汰最不活跃的会话
161 if (rx_sessions_.find(frame_start->frame_id) == rx_sessions_.end() && rx_sessions_.full()) {
162 EvictLeastRecentlyUsedSession();
163 }
164 auto &session = rx_sessions_[frame_start->frame_id];
165 session.payload_len = frame_start->payload_len;
166 session.payload_crc8 = frame_start->payload_crc8;
167 session.next_fragment_seq = 0;
168 session.last_active_time = std::chrono::steady_clock::now(); // 更新活跃时间
169 session.buffer.clear();
170 session.buffer.reserve(frame_start->payload_len);
171 } else if (msg->std_id == rx_stdid_base_ + StdIdOffset::kPayload) {
172 // 2. 收到数据分片
173 const auto *payload = reinterpret_cast<const decltype(tx_buf_.payload) *>(msg->data.data());
174 auto session_it = rx_sessions_.find(payload->frame_id);
175 if (session_it == rx_sessions_.end()) {
176 return; // 未找到对应会话，说明起始帧缺失
177 }
178 auto &session = session_it->second;
179 // 校验分片序号确保顺序正确
180 if (payload->fragment_seq != session.next_fragment_seq) {
181 rx_sessions_.erase(session_it); // 顺序错误，丢弃该会话
182 return;
183 }
184 const uint32_t data_len = payload->fragment_len;
185 if (session.buffer.size() + data_len > session.payload_len) {
186 rx_sessions_.erase(session_it); // 数据超长，丢弃该会话
187 return;
188 }
189 session.buffer.insert(session.buffer.end(), payload->fragment_data, payload->fragment_data + data_len);
190 session.next_fragment_seq++;
191 session.last_active_time = std::chrono::steady_clock::now(); // 更新活跃时间
192
193 } else if (msg->std_id == rx_stdid_base_ + StdIdOffset::kFrameEnd) {
194 // 3. 收到结束帧
195 const auto *frame_end = reinterpret_cast<const decltype(tx_buf_.frame_end) *>(msg->data.data());
196 auto session_it = rx_sessions_.find(frame_end->frame_id);
197 if (session_it == rx_sessions_.end()) {
198 return; // 没有对应的会话，忽略
199 }
200 auto &session = session_it->second;
201 if (session.buffer.size() != session.payload_len) {
202 rx_sessions_.erase(session_it); // 长度不一致，放弃接收
203 return;
204 }
205 // 4. 数据重组和CRC校验
206 uint8_t calculated_crc = modules::Crc8(session.buffer.empty() ? nullptr : session.buffer.data(),
207 session.buffer.size(), modules::CRC8_INIT);
208 if (calculated_crc == session.payload_crc8) {
209 // 5. CRC校验通过，调用回调函数处理完整的数据帧
210 if (frame_recv_callback_) {
211 std::string received_data(session.buffer.begin(), session.buffer.end());
212 frame_recv_callback_(received_data);
213 }
214 }
215 rx_sessions_.erase(session_it);
216 }
217 }

272 void EvictLeastRecentlyUsedSession() {
273 if (rx_sessions_.empty()) return;
274
275 auto oldest_it = rx_sessions_.begin();
276 auto oldest_time = oldest_it->second.last_active_time;
277
278 for (auto it = rx_sessions_.begin(); it != rx_sessions_.end(); ++it) {
279 if (it->second.last_active_time < oldest_time) {
280 oldest_time = it->second.last_active_time;
281 oldest_it = it;
282 }
283 }
284
285 rx_sessions_.erase(oldest_it);
286 }
