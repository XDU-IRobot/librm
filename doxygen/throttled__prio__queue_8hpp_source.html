45template <typename T, size_t MaxQueueSize>
46class ThrottledPrioQueue {
47 public:
48 using clock = std::chrono::steady_clock;
49 using time_point = clock::time_point;
50 using duration = clock::duration;
51
52 struct QueueItem {
53 T payload;
54 u8 priority;
55 time_point deadline;
56
57 // etl::priority_queue 默认是大顶堆，通过重载 < 决定谁在堆顶
58 bool operator<(const QueueItem& other) const {
59 if (priority == other.priority) {
60 // 优先级相同时，deadline 越早，越需要优先处理
61 // 因为是大顶堆，返回 true 会让 other 浮到堆顶
62 return deadline > other.deadline;
63 }
64 return priority < other.priority; // 优先级高的在堆顶
65 }
66 };
67
71 explicit ThrottledPrioQueue(double tx_frequency_hz)
72 : interval_(std::chrono::duration_cast<duration>(std::chrono::duration<double>(1.0 / tx_frequency_hz))) {}
73
81 bool Push(const T& payload, u8 priority, time_point deadline) {
82 if (queue_.full()) {
83 return false;
84 }
85 queue_.push({payload, priority, deadline});
86 return true;
87 }
88
93 std::optional<T> Process() { return Process(clock::now()); }
94
100 std::optional<T> Process(time_point now) {
101 // 1. 清理已超时的过期消息
102 while (!queue_.empty()) {
103 if (now >= queue_.top().deadline) {
104 queue_.pop();
105 } else {
106 break;
107 }
108 }
109
110 // 2. 检查队列是否为空或是否未到限频间隔
111 if (queue_.empty() || (now - last_process_time_ < interval_)) {
112 return std::nullopt;
113 }
114
115 // 3. 提取消息并更新状态
116 T payload = queue_.top().payload;
117 queue_.pop();
118 last_process_time_ = now;
119
120 return payload;
121 }
122
123 // 实用接口
124 size_t size() const { return queue_.size(); }
125 bool empty() const { return queue_.empty(); }
126 void Clear() {
127 while (!queue_.empty()) {
128 queue_.pop();
129 }
130 }
131
132 private:
133 etl::priority_queue<QueueItem, MaxQueueSize> queue_;
134 duration interval_;
135 time_point last_process_time_{};
136};
137} // namespace rm::modules
